# 《原地》
#
# 曲名取自全曲唯一一个不动的声部，不是气氛词：
# 丧钟从第 5 小节响到结束，始终只敲 C4 与 Gb3 两个音，一次没变；
# 而它下面那条长音从 C3–Gb3 一路沉到 Db2–G2，十一个半音。
# 所以听感上是**钟在原地，世界在往下掉** —— 也可以反过来读：
# 真正被困住的是那个不动的。两种读法都成立，这正是曲名要的歧义。
#
# 骨架：三全音长音 + 次低频压力 + 心跳 + 神经质刺音 + 丧钟。
# tempo 50 下 1 拍 = 1.2 秒，1 小节 = 4.8 秒，全曲 12 小节 = 57.6 秒
# （加上全局 9.0 秒混响留尾，导出总长约 66.6 秒）。
#
# 地基是一条逐小节下沉半音的三全音，十二小节走十一个半音：
#   1: C–Gb    2: B–F     3: Bb–E    4: A–Eb
#   5: Ab–D    6: G–Db    7: Gb–C    8: F–B
#   9: E–Bb   10: Eb–A   11: D–Ab   12: Db–G
# 三全音把八度正好切成两半，两个音谁也无法被听成主音 —— 没有一小节是稳定点，
# 所以 config 里的 key "Cm" 从头到尾没有被确认过一次。恐惧不来自不协和
# （三全音本身是干净的），来自**没有落点**。
#
# 另有一处不显眼的结构：第 7 小节（Gb2–C3）与第 1 小节（C3–Gb3）的音级完全相同，
# 都是 C 与 Gb —— 因为三全音对称，把 C3–Gb3 整体降一个三全音就得到 Gb2–C3。
# 更关键的是它与丧钟的对位：
#   - 第 5 小节丧钟敲下第一记 C4 时，长音在 Ab2–D3，两者斜着相撞；
#   - 第 7 小节丧钟敲下第二记 C4 时，长音恰好沉到 Gb2–C3 —— 钟的 C4 与长音顶部的 C3
#     锁成纯八度，全曲唯一一次"钟与地面合拢"，听上去像回到了开头，实则已沉下一层；
#   - 第 12 小节丧钟敲下最后一记 C4 时，长音沉到 Db2–G2 —— 同一个 C4 此刻悬在低音 Db2
#     上方复合大七度（差一个半音满两个八度），彻底失去落点。
#
# 四个阶段（声部有交叠，全部音序在 mix 中均以 loop=1 走完，无机械重复）：
#   第 1–4 小节   地基：长音、次低频压力、呼吸噪声、慢心跳；第 4 小节末心跳受惊早搏
#   第 3–6 小节   刺音插入 —— 刻意不落在段落边界上，惊吓感来自"没准备好"
#   第 5–8 小节   升级：丧钟进入，心跳在第 5–6 小节冲上每小节三组，第 7–8 小节透支回落
#   第 9–12 小节  崩塌：次低频压力撤掉（重量卸下去，只剩空的长音），
#                 心跳降回每小节一组、并逐小节往后挪，末小节只剩单独一记；
#                 第 11 小节丧钟的应答消失，末小节一记 C4 独自衰减 ——
#                 钟始终只敲 C4 与 Gb3，是它下面的世界沉了十一个半音
#
# 三段心跳（heartbeat_1 / heartbeat_2 / heartbeat_failing）是替换关系而非叠加，
# 且每段都是完整的 4 小节音序（loop=1）：第 1–3 小节平稳，第 3 小节末刺音扎入后，
# 第 4 小节立刻多出一次惊跳（早搏）接进第 5 小节的快心率；第 7–8 小节快心率开始漏拍、
# 掉音高，自然滑进第 9–12 小节的衰竭移位。十二小节是一条连续的生理曲线。
#
# 全部音序长度都是整小节数（4 / 8 / 12），mix 的 time 严格落在小节线上。

config {
  tempo: 50          # 慢到能听见"等待"，恐怖感主要来自这个速度
  master_gain: 0.7
  key: "Cm"          # 只是声明：全曲没有一个和弦能确认它，见文件头
  scale: "chromatic" # 长音走遍十二个半音，套任何调式都是假的，如实写"半音阶"
  swing: 0.12        # 极轻的摇摆，让偏拍的音站不稳
}

# 深渊长音：振幅 / 滤波 / 慢速 LFO 三层包络叠出"呼吸"的错觉。
# 每个和弦只有 1 小节（4.8 秒），所以 A+D 必须短于 4.8 秒 —— 否则引擎会按
# 比例压缩起音与衰减（见 applyAdsr）。这里取 2.0+2.5 = 4.5 < 4.8，包络按原值走。
# release 6.0 秒远长于一小节，相邻和弦因此始终交叠，中间不会出现断口。
#
# 加 velocity_to_filter: 160 是为了解决一个声学问题：当长音从 C3–Gb3 沉到 Db2–G2 时，
# 锯齿波基频降了近一个八度，它的二、三次谐波也会跟着落进 240Hz 低通带内；
# 若截止频率锁死，末段反而会比开头更亮、更毛躁。配合 drone_3 逐小节递减的 velocity，
# 滤波截止会随音高下沉同步收暗（第 12 小节压到约 195Hz），崩塌段才真正发空。
define_instrument(name="abyss_drone") {
  wave: "sawtooth"
  envelope: "adsr(2.0, 2.5, 0.85, 6.0, curve=linear)"
  filter: "lowpass(240, 2.0)"
  filter_envelope: "adsr(2.5, 2.0, 0.7, 5.0, curve=linear)"
  filter_env_amount: 180        # 加在 240Hz 基线上：2.5 秒开到 420Hz，再收到约 366Hz
  velocity_to_filter: 160       # 弱奏更暗：末段降低 velocity 时同步压低截止频率
  lfo: "sine(freq=0.13, amount=55, target=filter, ramp=4.0)"
  detune: 14
  voices: 3                     # 三层失谐锯齿：音墙而非单音
  unison_spread: 22             # 三个声部各偏 3 / 14 / 25 音分
  spread: 0.7                   # 声部左右铺开，低频不再挤在正中
  gain: 0.5
}

# 次低频压力：每个音都从下方五度滑上来，像胸腔被慢慢压住。
# glide_from 是相对目标音的半音数，-7 即"从纯五度下方滑入"。
# 每个音时值 2n = 2.4 秒：A+D = 1.2+1.0 = 2.2 < 2.4 秒，不触发压缩；
# LFO 的 ramp 设为 1.5 秒（< 2.4 秒），确保在单音时值内就能走完淡入。
# 这条音色只走第 1–8 小节，崩塌段撤掉它 —— 重量卸下去，只剩空的长音。
define_instrument(name="pressure") {
  wave: "sine"
  envelope: "adsr(1.2, 1.0, 0.85, 3.0, curve=linear)"
  glide: 0.9
  glide_from: -7
  gain: 0.6
  lfo: "sine(freq=0.22, amount=0.28, target=gain, ramp=1.5)"
}

# 神经质刺音：极短的 perc 包络 + 高 Q 带通 + 快速颤音，专做"抽气"式的惊吓
define_instrument(name="nerve_stab") {
  wave: "square"
  envelope: "perc(0.001, 0.08, 0.05)"
  filter: "bandpass(2400, 6)"
  filter_envelope: "perc(0.001, 0.06, 0.03)"
  filter_env_amount: 2800
  lfo: "sawtooth(freq=7.5, amount=45, target=frequency, ramp=0.02)"
  attack_noise: 0.3             # 起音处的噪声爆发：刺进来的那一下
  pan: -0.35
  gain: 0.5
  # 只留这一条延迟：刺音自己"追自己"，是这件音色的性格。
  # 混响交给全局链 —— 房间只有一个，不该每件乐器各写一份。
  effect_chain {
    delay(time="16n", feedback=0.38, mix=0.3, damping=2600)
  }
}

# 呼吸：粉噪声本身无音高，引擎对噪声波形挂带通（中心频率 = 音符频率，Q=1.2），
# 再叠加这里的 1200Hz 低通，让气流声随着音高逐小节下沉而变闷。
# breath_bed 中最短的音是 4n（1.2 秒），所以 A+D 设为 0.65+0.45 = 1.1 < 1.2 秒，
# LFO 的 ramp 设为 1.0 秒，保证最短的呼吸音也不会被引擎按比例压缩包络。
define_instrument(name="breath") {
  wave: "pink_noise"
  envelope: "adsr(0.65, 0.45, 0.78, 3.2, curve=linear)"
  filter: "lowpass(1200, 0.8)"
  lfo: "sine(freq=0.28, amount=0.35, target=gain, ramp=1.0)"
  gain: 0.35
  pan: -0.25
}

# 丧钟：钟体预设 + 显式长衰减包络 + 低通压暗。
# 为什么要显式写 envelope：preset "bell" 默认衰减较短且 sustain=0，
# 在 tempo 50 下 2n（2.4 秒）与末小节 1n（4.8 秒）听起来会一样长；
# 改用 3.6 秒衰减 + 0.08 持续电平 + 4.5 秒释音，末小节那一记 1n 才能真正把钟体余振拖满。
# 加 velocity_to_filter: 900 让应答音（Gb3，力度更轻）比主击（C4）更闷、更远。
define_instrument(name="toll") {
  preset: "bell"
  envelope: "adsr(0.003, 3.6, 0.08, 4.5)"
  filter: "lowpass(2400, 0.7)"
  velocity_to_filter: 900
  gain: 0.42
  pan: 0.3
  effect_chain {
    delay(time="4n.", feedback=0.42, mix=0.3, damping=1800)
  }
}

# 全局效果链：所有声部共用，顺序 = 信号流经顺序。
# 各乐器不再各自写混响，房间只有一个。
# eq 低频段在 250Hz、中频 1200Hz、高频 4000Hz：抬起 250Hz 以下、压暗中高音。
effect_chain {
  reverb(decay=9.0, mix=0.45, pre_delay=0.05, damping=2600)
  eq(low=1.5, mid=-1.0, high=-2.0)
}

# 长音分三段，每段 4 小节，接起来是一条连续下沉的三全音（见文件头对照表）。
# 拆三段是为了让"下沉"与其它声部（压力、心跳、丧钟）各自的进出点在 mix 中对齐小节线。
#
# 第 1–4 小节：从 C3–Gb3(130.8/185.0Hz) 起步，力度逐小节微涨（0.78 → 0.85），
# 把第 3 小节刺音扎入前后的压迫感慢慢顶起来。
sequence(name="drone_1", instrument="abyss_drone", gain=0.95) {
  chord(["C3",  "Gb3"], "1n", velocity=0.78)
  chord(["B2",  "F3"],  "1n", velocity=0.80)
  chord(["Bb2", "E3"],  "1n", velocity=0.82)
  chord(["A2",  "Eb3"], "1n", velocity=0.85)
}

# 第 5–8 小节：继续下沉。第 5–6 小节与丧钟、刺音正面碰撞，力度推到顶点（0.90）；
# 第 7 小节 Gb2–C3 与第 1 小节 C3–Gb3 音级完全相同（三全音对称），且顶部 C3 与
# 丧钟的 C4 锁成纯八度 —— 听上去像"回到开头"，实际比开头低了一个三全音；
# 第 8 小节力度开始回落（0.80），为第 9 小节抽走压力层做铺垫。
sequence(name="drone_2", instrument="abyss_drone", gain=0.95) {
  chord(["Ab2", "D3"],  "1n", velocity=0.88)
  chord(["G2",  "Db3"], "1n", velocity=0.90)
  chord(["Gb2", "C3"],  "1n", velocity=0.86)
  chord(["F2",  "B2"],  "1n", velocity=0.80)
}

# 第 9–12 小节：沉到 Db2–G2(69.3/98.0Hz)。这一段没有压力层托底，且 velocity
# 从 0.72 一路降到 0.52 —— 配合 abyss_drone 的 velocity_to_filter，滤波截止同步下压，
# 听起来不是变轻了，是脚下那块地板被抽走了。末小节 gate=1.35 让低音拖进混响尾音。
sequence(name="drone_3", instrument="abyss_drone", gain=0.95) {
  chord(["E2",  "Bb2"], "1n", velocity=0.72)
  chord(["Eb2", "A2"],  "1n", velocity=0.66)
  chord(["D2",  "Ab2"], "1n", velocity=0.60)
  chord(["Db2", "G2"],  "1n", velocity=0.52, gate=1.35)
}

# 次低频压力：与长音同一组音，但低一个八度，一小节两个音交替走三全音的两端。
# 长音是三层失谐锯齿，听感重心在中低音区；这一层是纯正弦，专门把最底下那一层补厚。
# 最低音是 F1 = 43.7Hz，用耳机能明显感受到胸腔压迫。
#
# 第 1–4 小节保持稳定推力；第 5–8 小节的后半段（第 7–8 小节）力度从 0.78 递减到 0.52，
# 让次低频在第 9 小节彻底消失前先有一个"往下漏空"的过程，而不是生硬切断。
sequence(name="pressure_1", instrument="pressure", gain=0.8) {
  note("C2",  "2n", velocity=0.80)
  note("Gb2", "2n", velocity=0.76)
  note("B1",  "2n", velocity=0.80)
  note("F2",  "2n", velocity=0.76)
  note("Bb1", "2n", velocity=0.82)
  note("E2",  "2n", velocity=0.78)
  note("A1",  "2n", velocity=0.84)
  note("Eb2", "2n", velocity=0.80)
}

sequence(name="pressure_2", instrument="pressure", gain=0.8) {
  note("Ab1", "2n", velocity=0.86)
  note("D2",  "2n", velocity=0.82)
  note("G1",  "2n", velocity=0.84)
  note("Db2", "2n", velocity=0.80)
  note("Gb1", "2n", velocity=0.78)
  note("C2",  "2n", velocity=0.72)
  note("F1",  "2n", velocity=0.64)
  note("B1",  "2n", velocity=0.52)
}

# 第 1–4 小节心跳（4 小节完整写出，不用 1 小节 loop=4）：
# 真实心音是"咚—哒"两下，第二声更闷更弱。
#   - 第 1–2 小节：每小节两组，平稳沉重；
#   - 第 3 小节：第 4 拍 nerve_scares 的第一记 C6 刺音突然扎入；
#   - 第 4 小节：心脏对刺音做出生理反应 —— 第二组提前半拍（只休止 8n），
#     并在小节末尾多挣出一记急促的早搏（tune=0），直接把心率顶进第 5 小节的快心率。
sequence(name="heartbeat_1") {
  # 第 1 小节：平稳（4 拍）
  hit("kick", "8n", tune=-4, decay=1.2,  snap=0.15, velocity=1.0)
  hit("kick", "8n", tune=-5, decay=0.7,  tone=-400, velocity=0.42)
  rest("4n")
  hit("kick", "8n", tune=-3, decay=1.05, snap=0.1,  velocity=0.8)
  hit("kick", "8n", tune=-6, decay=0.65, tone=-400, velocity=0.35)
  rest("4n")

  # 第 2 小节：微沉（4 拍）
  hit("kick", "8n", tune=-4, decay=1.2,  snap=0.15, velocity=0.95)
  hit("kick", "8n", tune=-5, decay=0.7,  tone=-400, velocity=0.40)
  rest("4n")
  hit("kick", "8n", tune=-4, decay=1.05, snap=0.1,  velocity=0.78)
  hit("kick", "8n", tune=-6, decay=0.65, tone=-400, velocity=0.34)
  rest("4n")

  # 第 3 小节：刺音在第 4 拍扎入，第二组心音略微绷紧（4 拍）
  hit("kick", "8n", tune=-3, decay=1.15, snap=0.15, velocity=0.95)
  hit("kick", "8n", tune=-5, decay=0.68, tone=-380, velocity=0.42)
  rest("4n")
  hit("kick", "8n", tune=-2, decay=1.0,  snap=0.12, velocity=0.85)
  hit("kick", "8n", tune=-4, decay=0.6,  tone=-380, velocity=0.38)
  rest("4n")

  # 第 4 小节：受惊后的早搏 —— 变成三组，末组音高上移，无缝接进第 5 小节（4 拍）
  hit("kick", "8n",  tune=-1, decay=0.95, snap=0.12, velocity=0.9)
  hit("kick", "8n",  tune=-4, decay=0.58, tone=-360, velocity=0.40)
  rest("8n")
  hit("kick", "8n",  tune=0,  decay=0.9,  snap=0.11, velocity=0.85)
  hit("kick", "8n",  tune=-3, decay=0.55, tone=-360, velocity=0.36)
  rest("8n")
  hit("kick", "8n",  tune=1,  decay=0.85, snap=0.1,  velocity=0.82)
  hit("kick", "16n", tune=-2, decay=0.5,  tone=-350, velocity=0.35)
  rest("16n")
}

# 第 5–8 小节心跳（4 小节完整写出，不用 loop=4）：
#   - 第 5–6 小节：丧钟进入 + 刺音后半段，每小节紧缩成三组（每组占 1 拍：8n+16n+16n），
#     音高上移、力度变薄 —— 心率快了，人却没力气了；
#   - 第 7 小节：刺音撤出，第三组开始体力透支（晚半拍才出来，音高掉回 0）；
#   - 第 8 小节：撑不住三组，退回两组且间隔拉大、音高转负，自然接进第 9 小节的单组衰竭。
sequence(name="heartbeat_2") {
  # 第 5 小节：紧迫三组（4 拍）
  hit("kick", "8n",  tune=2,  decay=0.85, snap=0.1,  velocity=0.82)
  hit("kick", "16n", tune=-1, decay=0.5,  tone=-350, velocity=0.36)
  rest("16n")
  hit("kick", "8n",  tune=2,  decay=0.85, snap=0.1,  velocity=0.76)
  hit("kick", "16n", tune=-1, decay=0.5,  tone=-350, velocity=0.33)
  rest("16n")
  hit("kick", "8n",  tune=3,  decay=0.8,  snap=0.08, velocity=0.72)
  hit("kick", "16n", tune=0,  decay=0.5,  tone=-350, velocity=0.30)
  rest("16n")
  rest("4n")

  # 第 6 小节：顶点三组，第三组已显虚弱（4 拍）
  hit("kick", "8n",  tune=3,  decay=0.82, snap=0.1,  velocity=0.80)
  hit("kick", "16n", tune=0,  decay=0.48, tone=-350, velocity=0.34)
  rest("16n")
  hit("kick", "8n",  tune=2,  decay=0.82, snap=0.09, velocity=0.74)
  hit("kick", "16n", tune=-1, decay=0.48, tone=-360, velocity=0.31)
  rest("16n")
  hit("kick", "8n",  tune=2,  decay=0.78, snap=0.08, velocity=0.68)
  hit("kick", "16n", tune=-1, decay=0.46, tone=-370, velocity=0.28)
  rest("16n")
  rest("4n")

  # 第 7 小节：第三组漏拍后移（休止 8n 再出），音高开始往下掉（4 拍）
  hit("kick", "8n",  tune=1,  decay=0.88, snap=0.09, velocity=0.76)
  hit("kick", "16n", tune=-2, decay=0.52, tone=-370, velocity=0.32)
  rest("16n")
  hit("kick", "8n",  tune=1,  decay=0.88, snap=0.08, velocity=0.70)
  hit("kick", "16n", tune=-2, decay=0.52, tone=-380, velocity=0.29)
  rest("16n")
  rest("8n")
  hit("kick", "8n",  tune=0,  decay=0.92, snap=0.08, velocity=0.64)
  hit("kick", "8n",  tune=-3, decay=0.55, tone=-390, velocity=0.27)
  rest("4n")
  rest("8n")

  # 第 8 小节：透支退回两组，音高落回负值，衔接到第 9 小节的衰竭（4 拍）
  hit("kick", "8n", tune=-1, decay=1.0,  snap=0.09, velocity=0.74)
  hit("kick", "8n", tune=-4, decay=0.58, tone=-400, velocity=0.30)
  rest("4n.")
  hit("kick", "8n", tune=-2, decay=1.05, snap=0.08, velocity=0.68)
  hit("kick", "8n", tune=-5, decay=0.58, tone=-410, velocity=0.28)
  rest("4n.")
}

# 第 9–12 小节：心率衰竭。每小节只剩一组"咚—哒"，
# 变的是两件事：它在小节里的位置逐小节往后挪，音高继续往下掉。
#   第 9 小节  第 1 拍起     第 10 小节  第 2 拍起
#   第 11 小节 第 3 拍起     第 12 小节  第 4 拍才起，第二声没跟上
# 用"位置后移"而不是"缩短时值"来表达停摆：心脏衰竭不是变快，
# 是每一次都来得比该来的晚一点。末小节那一记响完只剩半拍，全曲到此为止。
sequence(name="heartbeat_failing") {
  hit("kick", "8n", tune=-2, decay=1.1, snap=0.1, velocity=0.75)
  hit("kick", "8n", tune=-5, decay=0.6, tone=-420, velocity=0.3)
  rest("4n")
  rest("2n")

  rest("4n")
  hit("kick", "8n", tune=-3, decay=1.0, snap=0.08, velocity=0.65)
  hit("kick", "8n", tune=-6, decay=0.55, tone=-420, velocity=0.26)
  rest("2n")

  rest("2n")
  hit("kick", "8n", tune=-4, decay=0.9, snap=0.06, velocity=0.55)
  hit("kick", "8n", tune=-7, decay=0.5, tone=-440, velocity=0.22)
  rest("4n")

  rest("2n.")
  hit("kick", "8n", tune=-6, decay=0.8, snap=0.04, velocity=0.4)
  rest("8n")
}

# 12 小节：呼吸层铺满全曲，一次写完不靠 loop —— 它的音高决定噪声的带通中心
# （合成器对噪声波形挂的是 bandpass，中心频率 = 音符频率，Q=1.2），
# 所以要跟着长音一起往下沉，逐小节对应 drone_1/2/3 的十二个音。
# 每小节严格 4 拍，一小节一到两个音，其余留给空白。
sequence(name="breath_bed", instrument="breath", gain=0.75) {
  note("C4",  "2n", velocity=0.50)     # 1
  rest("4n")
  note("Gb3", "4n", velocity=0.44)

  note("B3",  "2n.", velocity=0.48)    # 2
  rest("4n")

  note("Bb3", "2n", velocity=0.46)     # 3
  rest("8n")
  note("E3",  "4n", velocity=0.40)
  rest("8n")

  note("A3",  "2n.", velocity=0.44)    # 4
  rest("4n")

  note("Ab3", "2n", velocity=0.42)     # 5
  rest("4n")
  note("D3",  "4n", velocity=0.38)

  note("G3",  "2n.", velocity=0.40)    # 6
  rest("4n")

  note("Gb3", "2n", velocity=0.38)     # 7
  rest("8n")
  note("C3",  "4n", velocity=0.34)
  rest("8n")

  note("F3",  "2n.", velocity=0.36)    # 8
  rest("4n")

  # 崩塌段：一小节只剩一个音，且都在低音区 —— 空间还在，人已经不在里面了
  note("E3",  "2n", velocity=0.32)     # 9
  rest("2n")

  note("Eb3", "2n", velocity=0.30)     # 10
  rest("2n")

  note("D3",  "2n.", velocity=0.28)    # 11
  rest("4n")

  note("Db3", "1n", velocity=0.26)     # 12：最后一个音拖满整小节，交给混响
}

# 4 小节，每小节严格 4 拍。大段留白 + 突然的刺音：
# 小二度簇与半音下行是最刺耳的两个动作，而中间的空才是主角 ——
# tempo 50 下一拍 1.2 秒，最短的 16n 只响 0.3 秒，每一击后面的静默都比它长得多。
#
# 五击刻意落在不同拍位（第 1 小节第 4 拍、第 2 小节第 2.5 拍、第 3 小节第 3.5 拍、
# 第 4 小节第 3 拍与第 4.5 拍），所以听不出周期 —— 恐怖片里最失效的就是
# "每 2 秒吓一次"，一旦可预测就不再吓人。
#
# 关于 run 的 direction：引擎里 `direction="down"` 会把数组反转，
# 所以要让音高**下行**，数组必须按上行写。这里写 ["A5","Bb5","B5","C6"]
# 加 direction="down"，实际发声是 C6→B5→Bb5→A5，与"半音下行"的意图一致。
sequence(name="nerve_scares", instrument="nerve_stab", gain=0.6) {
  rest("2n.")                                   # 第 1 小节（对应全曲第 3 小节）：静默 3 拍
  note("C6", "16n", velocity=1.0, accent=true)
  rest("8n")
  rest("16n")

  rest("4n.")                                   # 第 2 小节（对应全曲第 4 小节）：偏拍进入
  chord(["F5","Gb5"], "16n", velocity=0.9, pan=0.5)
  rest("2n")
  rest("16n")

  rest("2n")                                    # 第 3 小节（对应全曲第 5 小节）
  rest("8n")
  run(["A5","Bb5","B5","C6"], "32n", direction="down", velocity=0.75, pan=0.4)
  rest("4n")

  rest("2n")                                    # 第 4 小节（对应全曲第 6 小节）：先一声，再一簇
  note("Bb5", "8n", velocity=0.8, gate=0.6)
  rest("4n")
  chord(["C6","Db6"], "8n", velocity=1.0, accent=true, strum=0.15)
}

# 8 小节，每小节严格 4 拍。钟从头到尾只敲两个音：C4 与 Gb3 —— 正是全曲的
# 那个三全音。它自己不动，动的是它下面那条长音：从 Ab2–D3（第 5 小节）一路沉到
# Db2–G2（第 12 小节）。所以听感上不是钟在升，是钟在原地而世界在往下掉。
#
# 钟与刮擦是**先后**关系不是同时：音序内部的指令严格串行，每条把时间轴往后
# 推自己的时值，所以 hit() 出现在前一条 note() 之后，不会叠上去。
# 这里正好利用这一点做"一击一答"：C4 敲下（占 2 拍）→ 空 2 拍 → 刮擦 → Gb3 应答。
#
# 为什么钟用 C4 而不是与长音同八度的 C3：C4 = 261.6Hz，与长音整个音域都不重叠，
# 钟声才立得出来；而且在第 7 小节（长音 Gb2–C3）会与长音顶部的 C3 形成全曲唯一一次
# 纯八度对齐，到第 12 小节（长音 Db2–G2）则变成对低音 Db2 的复合大七度悬置。
sequence(name="bell_toll", instrument="toll", gain=0.85) {
  note("C4", "2n", velocity=0.9)        # 第 5 小节：第一记（底下是 Ab2–D3）
  rest("2n")

  hit("crash", "2n", decay=1.4, tone=-1800, velocity=0.2, pan=-0.6)
  note("Gb3", "2n", velocity=0.6)       # 第 6 小节：刮擦 + 应答（底下是 G2–Db3）

  note("C4", "2n", velocity=0.85)       # 第 7 小节：第二记（底下是 Gb2–C3，八度重合）
  rest("2n")

  hit("ride", "2n", decay=1.6, tone=-900, velocity=0.16, pan=0.5)
  note("Gb3", "2n", velocity=0.55)      # 第 8 小节（底下是 F2–B2）

  note("C4", "2n", velocity=0.8)        # 第 9 小节：第三记（底下是 E2–Bb2）
  rest("2n")

  hit("crash", "2n", decay=1.4, tone=-1800, velocity=0.18, pan=-0.5)
  note("Gb3", "2n", velocity=0.5)       # 第 10 小节（底下是 Eb2–A2）

  # 第 11 小节整小节留白 —— 前三次都是"一击一答"，到这里应答停了。
  # 此时底下长音是 D2–Ab2，与钟的 C–Gb 恰好拼成完整的减七和弦，钟却偏偏不响。
  rest("1n")

  # 第 12 小节：最后一记 C4（底下是 Db2–G2，悬在大七度上），没有应答，拖满整小节衰减进混响
  note("C4", "1n", velocity=0.75, gate=1.2)
}

# ===========================================================================
# 编排
# 全部 track 均为 loop=1 —— 全曲没有任何一段靠机械循环充数，
# 每个 4 小节阶段（包括心跳）都有自己独立的形态演变。
# 每条 track 的 time + 音序长度都落在 12 小节之内，全曲正好 57.6 秒。
# ===========================================================================
mix {
  # 第 1–4 小节：地基（第 4 小节心跳受第 3 小节刺音惊吓出现早搏）
  track(source="drone_1",           time=0,    loop=1, gain=0.9)
  track(source="pressure_1",        time=0,    loop=1, gain=0.85)
  track(source="heartbeat_1",       time=0,    loop=1, gain=0.95)

  # 第 3–6 小节：刺音刻意从第 3 小节插入，跨过第 4/5 小节边界
  track(source="nerve_scares",      time="2m", loop=1, gain=0.9)

  # 第 5–8 小节：升级（心率冲上三组后在第 7–8 小节透支回落，次低频压力在末两小节渐弱）
  track(source="drone_2",           time="4m", loop=1, gain=0.9)
  track(source="pressure_2",        time="4m", loop=1, gain=0.85)
  track(source="heartbeat_2",       time="4m", loop=1, gain=0.85)
  track(source="bell_toll",         time="4m", loop=1, gain=0.85)

  # 第 9–12 小节：崩塌。压力层撤掉，心跳换成逐小节后移的衰竭版
  track(source="drone_3",           time="8m", loop=1, gain=0.9)
  track(source="heartbeat_failing", time="8m", loop=1, gain=0.9)

  # 呼吸层铺满全曲，12 小节音高跟着长音逐小节下沉
  track(source="breath_bed",        time=0,    loop=1, gain=0.75, pan=-0.2)
}

# ---------------------------------------------------------------- 调参笔记
# 1. 恐怖音乐最忌讳声部同时换音。这里靠三件事错开：呼吸层的音符落在拍后或空拍，
#    心跳的第二声用更短的时值，以及刺音全部落在非重拍上。想让它更整齐反而会变得平淡。
#
# 2. 全曲所有 track 都是 loop=1：原稿中 heartbeat 与 heartbeat_fast 用 1 小节音序
#    配 loop=4，导致第 3 小节刺音扎进来时心跳毫无反应、第 8 小节到第 9 小节又从
#    三组直接断崖跳成一组。展开成 4 小节音序（heartbeat_1 / heartbeat_2）后，
#    第 4 小节末的早搏与第 7–8 小节的漏拍把十二小节串成了一条连续的生理曲线。
#
# 3. 包络与 LFO 时值核对：
#    - breath_bed 中最短的音是 4n（1.2 秒），因此 breath 的 A+D 设为 0.65+0.45 = 1.1 秒
#      （< 1.2 秒），LFO ramp 设为 1.0 秒，避免触发引擎的 A+D 按比例压缩。
#    - pressure 每个音是 2n（2.4 秒），A+D = 2.2 秒、LFO ramp = 1.5 秒，均在单音时值内完成。
#    - toll 显式声明 envelope: "adsr(0.003, 3.6, 0.08, 4.5)"，避免默认 bell 预设过早衰减，
#      让第 12 小节的 1n 长钟声真正拖满整小节并溢入 9 秒混响。
#
# 4. 声学补偿：abyss_drone 挂了 velocity_to_filter: 160，配合 drone_3 从 0.72 降到 0.52
#    的 velocity，在基频沉入 Db2–G2 时同步压低低通截止频率，防止高次谐波落入通带后反而变亮。
