config {
  tempo: 62
  master_gain: 0.62
}

// ---------------------------------------------------------------------------
// 主钢琴：1:1 FM 加厚度，衰减型包络，缓慢磁带漂移
// ---------------------------------------------------------------------------
define_instrument(name="warm_piano") {
  preset: "piano"
  wave: "sine"
  fm_wave: "sine"
  fm_ratio: 1.0            // 1:1 只产生谐波列，音准不跑
  fm_index: 75             // 单位是 Hz，对应 C4 约 0.29 调制指数：暖而不糊
  detune: 3

  envelope: adsr(0.005, 2.0, 0.16, 2.8, curve=exp)
  filter: lowpass(1200, 0.5)
  filter_envelope: adsr(0.008, 0.9, 0.25, 0.7)
  filter_env_amount: 1000  // 显式声明，控制起音的"开盖"幅度

  lfo: sine(freq=0.45, amount=3.5, target=frequency, ramp=2.0)  // ±3.5 音分慢漂移
  velocity_sensitivity: 0.85
  velocity_to_filter: 1600 // 强奏更亮
  gain: 0.8
  pan: -0.12
}

// ---------------------------------------------------------------------------
// 低频铺底：纯正弦地板，滤波不扫
// ---------------------------------------------------------------------------
define_instrument(name="subtle_pad") {
  wave: "sine"
  envelope: adsr(2.2, 1.2, 0.95, 5.0, curve=linear)
  filter: lowpass(320, 0.7)
  voices: 2
  unison_spread: 5
  spread: 0.35
  gain: 0.17
  pan: 0.0
}

// ---------------------------------------------------------------------------
// 玻璃钟：非整数比率 FM 产生钟声非谐分音
// ---------------------------------------------------------------------------
define_instrument(name="glass_bells") {
  preset: "bell"
  wave: "sine"
  fm_wave: "sine"
  fm_ratio: 3.5            // 非整数比 = 金属钟声
  fm_index: 260
  envelope: adsr(0.002, 1.3, 0.0, 1.9, curve=exp)   // 短促、有回响但不拖
  filter: highpass(280, 0.6)                        // 保住 G4(392) 及以上的基频
  voices: 2
  unison_spread: 6
  gain: 0.45
  pan: 0.3

  effect_chain {
    pingpong(time="4n", feedback=0.32, mix=0.3, damping=5000)
  }
}

// ---------------------------------------------------------------------------
// 全局效果链
// ---------------------------------------------------------------------------
effect_chain {
  eq(low=-2.5, mid=0.5, high=1.5)                          // 收低频泥浆，留高频光泽
  delay(time="8n.", feedback=0.26, mix=0.12, damping=2800)  // 62BPM 附点八分 = 0.726s
  reverb(decay=7.0, mix=0.4, pre_delay=0.04, damping=4000)  // 湿润大空间
}

// ===========================================================================
// 音序：全部 4 小节 = 15.48s，和声 Cm | Ab | Fm | G
// ===========================================================================

// 琶音：一小节一个和弦，四音分解（原谱音高与排列不变，时值 2n → 4n）
sequence(name="mid_range_arps", instrument="warm_piano", gain=0.6, humanize=0.16) {
  note("C2",  "4n", velocity=0.54); note("G2",  "4n", velocity=0.44)
  note("Eb3", "4n", velocity=0.46); note("C3",  "4n", velocity=0.42)

  note("Ab2", "4n", velocity=0.52); note("Eb3", "4n", velocity=0.44)
  note("C4",  "4n", velocity=0.46); note("Eb3", "4n", velocity=0.40)

  note("F2",  "4n", velocity=0.54); note("C3",  "4n", velocity=0.44)
  note("Ab3", "4n", velocity=0.46); note("C3",  "4n", velocity=0.42)

  note("G2",  "4n", velocity=0.56); note("D3",  "4n", velocity=0.46)
  note("F3",  "4n", velocity=0.48); note("D3",  "4n", velocity=0.44)
}

// 主旋律：连奏无缝隙（gate>1 让音符交叠，配合长 release 模拟踩住延音踏板）
sequence(name="main_legato_melody", instrument="warm_piano", gain=1.0, humanize=0.22) {
  note("G4",  "2n", velocity=0.78, gate=1.08)
  note("Eb4", "2n", velocity=0.68, gate=1.08)

  note("F4",  "4n", velocity=0.62)
  note("Eb4", "4n", velocity=0.70)
  note("D4",  "2n", velocity=0.74, gate=1.08)

  note("C4",  "2n", velocity=0.70, gate=1.08)
  note("Eb4", "2n", velocity=0.82, gate=1.08)   // 乐句高点

  note("D4",  "2n", velocity=0.74, gate=1.08)
  note("B3",  "2n", velocity=0.64, gate=1.10)   // 大调三音，构成 G7 解决回 Cm
}

// 高音对位：填主旋律留出的空隙，独立声像
sequence(name="sparkle_counterpoint", instrument="glass_bells", gain=0.85, humanize=0.26) {
  rest("4n")
  note("C5",  "8n", velocity=0.68, pan=0.24)
  note("D5",  "8n", velocity=0.62, pan=0.34)
  note("Eb5", "2n", velocity=0.80, pan=0.28)

  rest("2n")
  note("Bb4", "4n", velocity=0.60, pan=0.20)
  note("Ab4", "4n", velocity=0.66, pan=0.36)

  note("G4",  "8n", velocity=0.58, pan=0.22)
  note("C5",  "8n", velocity=0.68, pan=0.34)
  note("Eb5", "4n", velocity=0.76, pan=0.30)
  rest("2n")

  rest("2n")
  note("G5",  "8n", velocity=0.72, pan=0.40)
  note("F5",  "8n", velocity=0.64, pan=0.26)
  note("D5",  "4n", velocity=0.70, pan=0.32)
}

// 低频铺底：跟随和声根音
sequence(name="background_layer", instrument="subtle_pad", gain=0.9) {
  note("C3",  "1n", velocity=0.70)
  note("Ab2", "1n", velocity=0.64)
  note("F2",  "1n", velocity=0.68)
  note("G2",  "1n", velocity=0.72)
}

// ===========================================================================
// 编排：对位层第 2 轮才进入，前面先让钢琴与铺底立住
// ===========================================================================
mix {
  track(source="background_layer",     time=0,    loop=4)
  track(source="mid_range_arps",       time=0,    loop=4)
  track(source="main_legato_melody",   time=0,    loop=4)
  track(source="sparkle_counterpoint", time="4m", loop=3)
}
